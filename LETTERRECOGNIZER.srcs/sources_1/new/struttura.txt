library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity delay_module is
  generic (
    resolution : integer := 17;
    scale : integer := 8
  );
  port (
    clock : in std_logic;
    reset : in std_logic;
    start : in std_logic;
    delay_ms : in std_logic_vector( scale - 1 downto 0 );
    done : out std_logic
  );
end delay_module;

architecture behavioral of delay_module is
  signal low_counter : unsigned( resolution downto 0 );
  signal high_counter : unsigned( scale - 1 downto 0 );
  signal count_init, count_en : std_logic;
  signal local_delay : unsigned( scale - 1 downto 0 );
  signal load_en : std_logic;
  type state_t is ( idle, count, timesup );
  signal pstate, nstate : state_t;
begin
  process ( clock, reset ) begin
    if ( reset = '0' ) then
      pstate <= idle;
    elsif rising_edge( clock ) then
      pstate <= nstate;
    end if;
  end process;

  process ( pstate, start, local_delay, high_counter ) begin
    nstate <= pstate;
    count_init <= '0';
    count_en <= '0';
    done <= '0';
    load_en <= '0';
    case pstate is
      when idle =>
        count_init <= '1';
        if ( start = '1' ) then
          load_en <= '1';
          nstate <= count;
        end if;
      when count =>
        count_en <= '1';
        if ( local_delay = high_counter ) then
          nstate <= timesup;
        end if;
      when timesup =>
        done <= '1';
        nstate <= idle;
    end case;
  end process;

  process ( clock, reset ) begin
    if ( reset = '0' ) then
      local_delay <= to_unsigned( 0, scale );
    elsif rising_edge( clock ) then
      if ( load_en = '1' ) then
        local_delay <= unsigned( delay_ms );
      end if;
    end if;
  end process;

  process ( clock, reset ) begin
    if ( reset = '0' ) then
      low_counter <= to_unsigned( 0, resolution + 1 );
      high_counter <= to_unsigned( 0, scale );
    elsif rising_edge( clock ) then
      if ( count_init = '1' ) then
        low_counter <= to_unsigned( 0, resolution + 1 );
        high_counter <= to_unsigned( 0, scale );
      elsif ( count_en = '1' ) then
        if ( low_counter( resolution ) = '1' ) then
          low_counter <= to_unsigned( 0, resolution + 1 );
          high_counter <= high_counter + 1;
        else
          low_counter <= low_counter + 1;
        end if;
      end if;
    end if;
  end process;

end behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;
use work.mlp_weights.ALL;

entity mlp is
    port (
        clk       : in  std_logic;
        reset     : in  std_logic;
        start     : in  std_logic;
        input     : in  std_logic_vector(7 downto 0);
        input_addr: out integer range 0 to 195;
        done      : out std_logic;
        result    : out std_logic_vector(4 downto 0)
    );
end mlp;

architecture Behavioral of mlp is
    type state_t is (IDLE, LOAD_INPUT, COMPUTE_HIDDEN, COMPUTE_OUTPUT, ARGMAX, DONE_ST);
    signal state : state_t := IDLE;

    type input_array is array (0 to 195) of integer range 0 to 1;
    signal input_buf : input_array := (others => 0);

    type hidden_array is array (0 to 31) of integer;
    signal hidden_buf : hidden_array := (others => 0);

    type output_array is array (0 to 25) of integer;
    signal output_buf : output_array := (others => 0);

    -- NUOVO: array di registri per accumulare i valori in parallelo
    type acc_hidden_t is array (0 to 31) of integer;
    signal acc_hidden : acc_hidden_t := (others => 0);

    type acc_output_t is array (0 to 25) of integer;
    signal acc_output : acc_output_t := (others => 0);

    signal idx : integer range 0 to 255 := 0;
    signal jdx : integer range 0 to 255 := 0;

    signal max_val : integer := -2147483648;
    signal max_idx : integer range 0 to 25 := 0;
begin

    input_addr <= idx when state = LOAD_INPUT else 0;

    process(clk, reset)
    begin
        if reset = '1' then
            state <= IDLE;
            done <= '0';
            result <= (others => '0');
            idx <= 0;
            jdx <= 0;
            acc_hidden <= (others => 0);
            acc_output <= (others => 0);
            max_val <= -2147483648;
            max_idx <= 0;
        elsif rising_edge(clk) then
            case state is
                when IDLE =>
                    done <= '0';
                    if start = '1' then
                        state <= LOAD_INPUT;
                        idx <= 0;
                    end if;

                when LOAD_INPUT =>
                    if to_integer(unsigned(input)) > 0 then
                        input_buf(idx) <= 1;
                    else
                        input_buf(idx) <= 0;
                    end if;
                    
                    if idx = 195 then
                        state <= COMPUTE_HIDDEN;
                        idx <= 0;
                        acc_hidden <= (others => 0); -- Prepara accumulatori a 0
                    else
                        idx <= idx + 1;
                    end if;

                when COMPUTE_HIDDEN =>
                    if idx < 196 then
                        -- PARALLELIZZAZIONE: Genera 32 operazioni MAC fisicamente separate
                        -- che avvengono tutte nello stesso istante (ciclo di clock).
                        for i in 0 to 31 loop
                            acc_hidden(i) <= acc_hidden(i) + input_buf(idx) * W1(idx * 32 + i);
                        end loop;
                        idx <= idx + 1;
                    else
                        -- Applica Bias e ReLU simultaneamente per tutti i 32 neuroni
                        for i in 0 to 31 loop
                            if acc_hidden(i) + B1(i) < 0 then
                                hidden_buf(i) <= 0;
                            else
                                hidden_buf(i) <= acc_hidden(i) + B1(i);
                            end if;
                        end loop;
                        state <= COMPUTE_OUTPUT;
                        idx <= 0;
                        acc_output <= (others => 0);
                    end if;

                when COMPUTE_OUTPUT =>
                    if idx < 32 then
                        -- PARALLELIZZAZIONE: Genera 26 operazioni MAC simultanee
                        for i in 0 to 25 loop
                            acc_output(i) <= acc_output(i) + hidden_buf(idx) * W2(idx * 26 + i);
                        end loop;
                        idx <= idx + 1;
                    else
                        -- Applica Bias finale simultaneamente per i 26 neuroni
                        for i in 0 to 25 loop
                            output_buf(i) <= acc_output(i) + B2(i);
                        end loop;
                        state <= ARGMAX;
                        jdx <= 0;
                        max_val <= -2147483648;
                        max_idx <= 0;
                    end if;

                when ARGMAX =>
                    -- Manteniamo l'ArgMax sequenziale (26 cicli sono trascurabili)
                    if jdx < 26 then
                        if output_buf(jdx) > max_val then
                            max_val <= output_buf(jdx);
                            max_idx <= jdx;
                        end if;
                        jdx <= jdx + 1;
                    else
                        result <= std_logic_vector(to_unsigned(max_idx, 5));
                        state <= DONE_ST;
                    end if;

                when DONE_ST =>
                    done <= '1';
                    state <= IDLE;
            end case;
        end if;
    end process;

end Behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity oled_counter is
  generic (
    -- Size in bits of the counter
    SIZE : integer := 4;
    -- Value at which the terminal count is asserted
    TERMINAL : integer := 15
  );
  port (
    -- System clock
    clock : in std_logic;
    -- Global reset (active low)
    reset : in std_logic;
    -- Count init
    counter_init : in std_logic;
    -- Count enable
    counter_enable : in std_logic;
    -- Terminal count
    counter_tc : out std_logic;
    -- Count value
    counter_val : out unsigned( SIZE - 1 downto 0 )
  );
end oled_counter;

architecture behavioral of oled_counter is

  -- Temporary value to hold the count
  signal the_count : unsigned( SIZE - 1 downto 0 );

begin

  counter_val <= the_count;

  process ( clock, reset ) begin
    if reset = '0' then
      the_count <= ( others => '0' );
    elsif rising_edge( clock ) then
      if ( counter_init = '1' ) then
        the_count <= ( others => '0' );
      elsif ( counter_enable = '1' ) then
        the_count <= the_count + 1;
      end if;
    end if;
  end process;

  counter_tc <= '1' when ( the_count = TERMINAL ) else '0';

end behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity spi_controller is
  port (
    clock : in std_logic;
    reset : in std_logic;
    start : in std_logic;
    data_in : in std_logic_vector( 7 downto 0 );
    sdata : out std_logic;
    sclk : out std_logic;
    latch : out std_logic;
    done : out std_logic
  );
end spi_controller;

architecture behavioral of spi_controller is
  type state_t is ( idle, transmit );
  signal pstate, nstate : state_t;
  signal psclk, nsclk : std_logic;
  signal clk_counter : unsigned( 3 downto 0 );
  signal clk_count_en : std_logic;
  signal clk_tc : std_logic;
  signal bit_counter : unsigned( 2 downto 0 );
  signal bit_count_en : std_logic;
  signal bit_tc : std_logic;
  signal count_init : std_logic;
  signal shift_register : std_logic_vector( 7 downto 0 );
  signal load_en : std_logic;
  signal shift_en : std_logic;
begin
  sclk <= psclk;

  process ( clock, reset ) begin
    if ( reset = '0' ) then
      pstate <= idle;
      psclk <= '1';
    elsif ( rising_edge( clock ) ) then
      pstate <= nstate;
      psclk <= nsclk;
    end if;
  end process;

  process ( pstate, psclk, start, clk_tc, bit_tc ) begin
    nstate <= pstate;
    nsclk <= psclk;
    count_init <= '0';
    load_en <= '0';
    latch <= '0';
    done <= '0';
    clk_count_en <= '0';
    bit_count_en <= '0';
    shift_en <= '0';
    case pstate is
      when idle =>
        count_init <= '1';
        done <= '0';
        if ( start = '1' ) then
          nstate <= transmit;
          nsclk <= '0';
          load_en <= '1';
          latch <= '1';
          done <= '0';
        end if;
      when transmit =>
        clk_count_en <= '1';
        case psclk is
          when '0' =>
            if ( clk_tc = '1' ) then
              nsclk <= '1';
            end if;
          when '1' =>
            if ( clk_tc = '1' ) then
              shift_en <= '1';
              if ( bit_tc = '0' ) then
                bit_count_en <= '1';
                nsclk <= '0';
              else
                nstate <= idle;
                done <= '1';
              end if;
            end if;
          when others =>
        end case;
    end case;
  end process;

  process ( clock, reset ) begin
    if ( reset = '0' ) then
      shift_register <= ( others => '0' );
    elsif ( rising_edge( clock ) ) then
      if ( load_en = '1' ) then
        shift_register <= data_in;
      elsif ( shift_en = '1' ) then
        shift_register <= shift_register( 6 downto 0 ) & '0';
      end if;
    end if;
  end process;
  sdata <= shift_register( 7 );

  process ( clock, reset ) begin
    if ( reset = '0' ) then
      clk_counter <= ( others => '1' );
    elsif ( rising_edge( clock ) ) then
      if ( count_init = '1' ) then
        clk_counter <= ( others => '1' );
      elsif ( clk_count_en = '1' ) then
        clk_counter <= clk_counter - 1;
      end if;
    end if;
  end process;
  clk_tc <= '1' when clk_counter = 0 else '0';

  process ( clock, reset ) begin
    if ( reset = '0' ) then
      bit_counter <= ( others => '1' );
    elsif ( rising_edge( clock ) ) then
      if ( count_init = '1' ) then
        bit_counter <= ( others => '1' );
      elsif ( bit_count_en = '1' ) then
        bit_counter <= bit_counter - 1;
      end if;
    end if;
  end process;
  bit_tc <= '1' when bit_counter = 0 else '0';

end behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity uart_rx is
    generic (
        CLK_FREQ : integer := 25_000_000;
        BAUD_RATE : integer := 115200
    );
    port (
        clk       : in  std_logic;
        reset     : in  std_logic;
        rx        : in  std_logic;
        data_out  : out std_logic_vector(7 downto 0);
        data_valid: out std_logic
    );
end uart_rx;

architecture Behavioral of uart_rx is
    constant BIT_PERIOD : integer := CLK_FREQ / BAUD_RATE;
    constant HALF_BIT   : integer := BIT_PERIOD / 2;
    type state_t is (IDLE, START, DATA, STOP);
    signal state : state_t := IDLE;
    signal bit_cnt : integer range 0 to 7 := 0;
    signal clk_cnt : integer range 0 to BIT_PERIOD-1 := 0;
    signal rx_shift : std_logic_vector(7 downto 0) := (others => '0');
    signal rx_sync : std_logic_vector(1 downto 0) := (others => '1');
begin
    process(clk)
    begin
        if rising_edge(clk) then
            rx_sync <= rx_sync(0) & rx;
        end if;
    end process;

    process(clk, reset)
    begin
        if reset = '1' then
            state <= IDLE;
            bit_cnt <= 0;
            clk_cnt <= 0;
            data_valid <= '0';
        elsif rising_edge(clk) then
            data_valid <= '0';
            case state is
                when IDLE =>
                    if rx_sync(1) = '0' then
                        state <= START;
                        clk_cnt <= 0;
                    end if;
                when START =>
                    if clk_cnt = HALF_BIT then
                        if rx_sync(1) = '0' then
                            state <= DATA;
                            clk_cnt <= 0;
                            bit_cnt <= 0;
                        else
                            state <= IDLE;
                        end if;
                    else
                        clk_cnt <= clk_cnt + 1;
                    end if;
                when DATA =>
                    if clk_cnt = BIT_PERIOD - 1 then
                        clk_cnt <= 0;
                        rx_shift(bit_cnt) <= rx_sync(1);
                        if bit_cnt = 7 then
                            state <= STOP;
                        else
                            bit_cnt <= bit_cnt + 1;
                        end if;
                    else
                        clk_cnt <= clk_cnt + 1;
                    end if;
                when STOP =>
                    if clk_cnt = BIT_PERIOD - 1 then
                        state <= IDLE;
                        data_out <= rx_shift;
                        data_valid <= '1';
                    else
                        clk_cnt <= clk_cnt + 1;
                    end if;
            end case;
        end if;
    end process;
end Behavioral;

entity uart_tx is

        BAUD_RATE : integer := 115200
    );

        start   : in  std_logic;
        tx      : out std_logic;
        busy    : out std_logic
    );
end uart_tx;

architecture Behavioral of uart_tx is
    constant BIT_PERIOD : integer := CLK_FREQ / BAUD_RATE;
    -- RINOMINATO START -> START_ST PER EVITARE CONFLITTO CON LA PORTA "start"
    type state_t is (IDLE, START_ST, DATA, STOP);
    signal state : state_t := IDLE;
    signal bit_cnt : integer range 0 to 7 := 0;
    signal clk_cnt : integer range 0 to BIT_PERIOD-1 := 0;
    signal tx_shift : std_logic_vector(7 downto 0) := (others => '0');
begin
    process(clk, reset)
    begin
        if reset = '1' then
            state <= IDLE;
            tx <= '1';
            busy <= '0';
        elsif rising_edge(clk) then
            case state is
                when IDLE =>
                    tx <= '1';
                    busy <= '0';
                    if start = '1' then
                        state <= START_ST;
                        tx_shift <= data_in;
                        clk_cnt <= 0;
                        busy <= '1';
                    end if;
                when START_ST =>
                    tx <= '0';
                    if clk_cnt = BIT_PERIOD-1 then
                        clk_cnt <= 0;
                        state <= DATA;
                        bit_cnt <= 0;
                    else
                        clk_cnt <= clk_cnt + 1;
                    end if;
                when DATA =>
                    tx <= tx_shift(bit_cnt);
                    if clk_cnt = BIT_PERIOD-1 then
                        clk_cnt <= 0;
                        if bit_cnt = 7 then
                            state <= STOP;
                        else
                            bit_cnt <= bit_cnt + 1;
                        end if;
                    else
                        clk_cnt <= clk_cnt + 1;
                    end if;
                when STOP =>
                    tx <= '1';
                    if clk_cnt = BIT_PERIOD-1 then
                        state <= IDLE;
                    else
                        clk_cnt <= clk_cnt + 1;
                    end if;
            end case;
        end if;
    end process;
end Behavioral;
